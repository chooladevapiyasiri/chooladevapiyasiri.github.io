  region,
  date_trunc('month', order_date) AS month,
  SUM(margin) AS total_margin,
  SUM(margin) - LAG(SUM(margin)) OVER (
    PARTITION BY region ORDER BY date_trunc('month', order_date)
  ) AS margin_change
FROM orders
GROUP BY region, month
ORDER BY region, month;
